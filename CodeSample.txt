   public function student_view($student_id,$page_name='account')
	 {
		if(!$student_id)
			return redirect()->to(base_url().'/admissions'); 	
			
		$student_info       = $this->AdmissionsModel->admission_detail_info($student_id);	
		$billing_info       = $this->AdmissionsModel->student_billing_info($student_id);	
		$notification_info  = $this->AdmissionsModel->student_notification_info($student_id);
		$student_slots_info = $this->AdmissionsModel->student_slots_info($student_id);
		
		$student_comments   = $this->AdmissionsModel->get_studnet_comments($student_id);
		$studnet_documents  = $this->AdmissionsModel->get_studnet_documents($student_id);
		$studnet_pmt_meetings = $this->AdmissionsModel->get_studnet_pmt_meetings($student_id);
		
	
		$data = array('student_info' => $student_info,'page_name' => $page_name,'student_id' => $student_id,'rooms_sessions_lists'=>rooms_sessions_lists(),
		              'billing_info' => $billing_info,'notification_info' => $notification_info,
		              'student_slots_info' => $student_slots_info,'AllTeacherssDropdown'=>$this->AdmissionsModel->AllTeacherssDropdown(),
		              'all_doctors'  => $this->AdmissionsModel->AllDoctorsDropdown(),'student_comments'=>$student_comments,
		              'student_documents'=>$studnet_documents,'studnet_pmt_meetings'=>$studnet_pmt_meetings
		              );	
		if($page_name=='sessions'){
		   return view('admissions/student_sessions',$data);	  
		   }
		else if($page_name=='billing'){
		    
		    
		   return view('admissions/student_billing',$data);	  
		   } 
		else if($page_name=='notifications'){
		    
		    
		   return view('admissions/student_notifications',$data);	  
		   }    
		else{
		    return view('admissions/student_view',$data);	  
		   }	
	 }

	function bulk_fee_entries($fee_counter,$admision_res,$MonthlyFee,$is_paid){
	    
	    for($i=$fee_counter;$i<=12;$i++){
			 $today_date      = date('Y-m-d');
			 $payment_rcvd_on = date('Y-m-d H:i:s');
			 $time            = strtotime(date('Y-m-d'));
			 $due_date        =  date("Y-m-d", strtotime("+".$i." month", $time));
         	 $payment_rcvd_on = '';
			 $insert_installment_data = array("admission_id"=>$admision_res,"fee_amount"=>$MonthlyFee,"is_paid"=>$is_paid,
					                          "due_date"=>$due_date,"payment_rcvd_on"=>$payment_rcvd_on
					                                     ); 
			$this->AdmissionsModel->add_admission_fee($insert_installment_data);
					        
		}  
	    
	}

   public function mark_left($id)
	{	 
	    $DeleteRec = $this->AdmissionsModel->markleftRecord($id);
		if($DeleteRec){
			$this->message_output->set_success('Record marked left successfully');
		} else {
			$this->message_output->set_error( 'Please try again! Error while deleting record.');
		}
		return redirect()->to(base_url().'/admissions'); 	
	}
}
